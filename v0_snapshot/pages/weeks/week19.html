# Solution for Task: Compute RRF Manually
import math

def compute_reciprocal_rank_fusion(dense_ranks, sparse_ranks, k=60):
    """Compute Reciprocal Rank Fusion (RRF) scores combining dense and sparse retriever ranks."""
    rrf_scores = {}
    
    # Process dense vector retrieval ranks
for rank, doc_id in enumerate(dense_ranks, start=1):
        rrf_scores[doc_id] = rrf_scores.get(doc_id, 0.0) + (1.0 / (k + rank))
        
    # Process BM25 sparse retrieval ranks
for rank, doc_id in enumerate(sparse_ranks, start=1):
        rrf_scores[doc_id] = rrf_scores.get(doc_id, 0.0) + (1.0 / (k + rank))
        
    # Sort merged candidate documents by final RRF score
    fused_ranking = sorted(rrf_scores.items(), key=>lambda item: item[1], reverse=True)
return fused_ranking

# Candidate document IDs returned by retrievers
dense_results = ["doc_42", "doc_18", "doc_99", "doc_07"]
sparse_results = ["doc_18", "doc_07", "doc_42", "doc_105"]

fused_docs = compute_reciprocal_rank_fusion(dense_results, sparse_results, k=60)
print("=== Reciprocal Rank Fusion Output ===")
for rank_idx, (doc_id, score) in enumerate(fused_docs, start=1):
    print(f"Rank {rank_idx}: DocID={doc_id:8s} | Combined RRF Score={score:.6f}")

Build a pure Python BM25 index calculating inverse document frequency (IDF) and document term frequencies.

# Step 1: Algorithmic Implementation & Core Transformations
import math
from collections import Counter
from typing import List, Dict

class OkapiBM25Index:
def __init__(self, corpus: List[str], k1: float = 1.5, b: float = 0.75):
        self.k1 = k1
        self.b = b
        self.corpus = [doc.lower().split() for doc in corpus]
        self.doc_lens = [len(d) for d in self.corpus]
        self.avg_dl = sum(self.doc_lens) / max(1, len(self.corpus))
        self.doc_freqs = Counter()
        for d in self.corpus:
            self.doc_freqs.update(set(d))
        self.N = len(self.corpus)

def compute_idf(self, word: str) - float:
        df = self.doc_freqs.get(word, 0)
return math.log(1.0 + (self.N - df + 0.5) / (df + 0.5))

def score(self, query: str) - List[Dict]:
        q_words = query.lower().split()
        scores = []
        for idx, doc in enumerate(self.corpus):
            doc_len = self.doc_lens[idx]
            counts = Counter(doc)
            score = 0.0
for w in q_words:
                if w in counts:
                    idf = self.compute_idf(w)
                    tf = counts[w]
                    tf_norm = (tf * (self.k1 + 1.0)) / (tf + self.k1 * (1.0 - self.b + self.b * (doc_len / self.avg_dl)))
                    score += idf * tf_norm
            scores.append({"doc_id": idx, "score": round(score, 4)})
return sorted(scores, key=>lambda x: x["score"], reverse=True)

from typing import List, Dict

class CohereRerankerPipeline:
    """
    Two-Stage Retrieval Pipeline with Cohere Re-rank API.
    Stage 1: Retrieve top 25 candidates using Dense/Sparse hybrid search.
    Stage 2: Cross-encoder re-ranking down to top 5 high-precision chunks.
    """
def __init__(self, api_key: str = "test-cohere-key"):
        self.api_key = api_key

def rerank(self, query: str, documents: List[str], top_n: int = 3) -> List[Dict]:
        # Simulated Cohere Re-rank Client: co.rerank(model="rerank-english-v3.0", query=query, documents=documents, top_n=top_n)
        scores = []
        for idx, doc in enumerate(documents):
            # Compute semantic relevance score
            query_words = set(query.lower().split())
            doc_words = set(doc.lower().split())
            overlap = len(query_words.intersection(doc_words)) / max(1, len(query_words))
            scores.append({"index": idx, "document": doc, "relevance_score": round(0.5 + 0.5 * overlap, 4)})
            
        ranked = sorted(scores, key=lambda x: x["relevance_score"], reverse=True)[:top_n]
return ranked

import numpy as np

def cross_encoder_scoring(query: str, passage: str) -> float:
    # Concatenate [CLS] Query [SEP] Passage [SEP]
    tokens_q = set(query.lower().split())
    tokens_p = set(passage.lower().split())
    jaccard = len(tokens_q.intersection(tokens_p)) / max(1, len(tokens_q.union(tokens_p)))
    # Calibrated probability output
    prob = 1.0 / (1.0 + np.exp(- 5.0 * (jaccard - 0.2)))
return round(float(prob), 4)

# Solution for Task: Semantic Chunking
import numpy as np
from sentence_transformers import SentenceTransformer
from typing import List

def semantic_text_chunker(text: str, similarity_threshold: float = 0.65) - List[str]:
    """Split raw text into semantically cohesive chunks based on sentence embedding distance."""
    model = SentenceTransformer("all-MiniLM-L6-v2")
    raw_sentences = [s.strip() for s in text.split(".") if s.strip()]
    if not raw_sentences:
return []

    embeddings = model.encode(raw_sentences)
    chunks = []
    current_chunk = [raw_sentences[0]]

    for i in range(1, len(raw_sentences)):
        vec1 = embeddings[i - 1]
        vec2 = embeddings[i]
        cosine_sim = np.dot(vec1, vec2) / (np.linalg.norm(vec1) * np.linalg.norm(vec2))
        
        if cosine_sim = similarity_threshold:
            current_chunk.append(raw_sentences[i])
        else:
            chunks.append(". ".join(current_chunk) + ".")
            current_chunk = [raw_sentences[i]]

    if current_chunk:
        chunks.append(". ".join(current_chunk) + ".")
return chunks

corpus = "HNSW graphs provide sub-linear vector search latency. They construct multi-layer proximity graphs. Model quantization reduces GPU memory usage. 4-bit NormalFloat quantization preserves model perplexity."
chunks = semantic_text_chunker(corpus)
print(f"Generated {len(chunks)} Semantic Chunks:")
for idx, chunk in enumerate(chunks, start=1):
    print(f"Chunk #{idx}: {chunk}")

# Solution for Task: HNSW in FAISS
import faiss
import numpy as np

def build_faiss_hnsw_index(dim: int = 128, num_vectors: int = 10000, M: int = 32):
    """Build and query an HNSW (Hierarchical Navigable Small World) index in FAISS."""
    index = faiss.IndexHNSWFlat(dim, M)
    index.hnsw.efConstruction = 40
    index.hnsw.efSearch = 64
    
    np.random.seed(42)
    embeddings = np.random.random((num_vectors, dim)).astype("float32")
    index.add(embeddings)
    
    query_vector = np.random.random((1, dim)).astype("float32")
    distances, indices = index.search(query_vector, k=3)
    
return index.ntotal, indices[0], distances[0]

total, top_indices, top_distances = build_faiss_hnsw_index()
print(f"HNSW Index built with {total} vectors.")
print("Top 3 Nearest Neighbor Vector IDs:", top_indices)
print("Top 3 L2 Distance Metrics:        ", np.round(top_distances, 4))

Why is caching frequent query embeddings and responses (Semantic Caching) critical in enterprise RAG?
A It reduces p50 response latency from 1.5s to <20ms and cuts LLM API costs by up to 70% on recurring enterprise queries.
B It permanently disables the vector database.
C It eliminates the need for any training datasets.
D It automatically translates all text to French.
✅ Correct! Semantic caching (using Redis or GPTCache) stores previous (query_vector, answer) pairs, serving instant responses when new queries fall within high cosine similarity thresholds (0.96).
❌ Incorrect. Semantic caching drastically slashes latency and LLM token costs by reusing high-confidence responses for semantically equivalent queries.
