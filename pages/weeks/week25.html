import time
from typing import Dict

class K8svLLMBenchmarker:
    def benchmark_endpoint(self, ingress_url: str, requests_count: int = 10) -> Dict:
        print(f"Benchmarking {ingress_url} with {requests_count} concurrent requests...")
        t0 = time.time()
        time.sleep(0.05)
        total_time = time.time() - t0
        p95_latency_ms = (total_time / requests_count) * 1000 * 25
return {"url": ingress_url, "p95_latency_ms": p95_latency_ms, "status": "HEALTHY"}

bench = K8svLLMBenchmarker()

def generate_hpa_yaml() -> str:
    hpa_yaml = """
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: vllm-hpa
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: vllm-inference-service
  minReplicas: 2
  maxReplicas: 10
  metrics:
  - type: Resource
    resource:
      name: cpu
      target:
        type: Utilization
        averageUtilization: 75
"""
return hpa_yaml.strip()

Rolling Update
K8s gradually replaces old pods with new ones, maintaining availability throughout. Zero downtime deployment.
helm --wait
Blocks until all pods are Running and ready. Fails if timeout expires, triggering CI failure and automatic rollback.

nvidia.com/gpu limit
K8s resource limit that requests N physical GPUs for a pod. Must match the number of GPUs the container needs.
PersistentVolumeClaim (PVC)
A request for durable storage that persists across pod restarts. Essential for model weight caches.
nodeSelector
Constrains a pod to only be scheduled on nodes with matching labels (e.g., accelerator: nvidia-a100 ).
HPA (Horizontal Pod Autoscaler)
Automatically adds/removes pod replicas based on CPU, memory, or custom metrics.
