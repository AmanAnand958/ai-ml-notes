# Build a Stats Calculator from Scratch
import numpy as np
def describe_series(data):
    a = np.array(data)
    m = np.mean(a)
    md = np.median(a)
    v = np.var(a,ddof=1)
    s = np.std(a,ddof=1)
    q1,q3 = np.percentile(a,[25,75])
    iqr=q3-q1
    n=len(a)
    sk = (n*np.sum((a-m)**3)/((n-1)*(n-2)*s**3)) if n>2 else 0
    print(f"{'Stat':<12} {'Value':<10}")
    print("-"*22)
    for l,v in [("Mean",m),("Median",md),("Var",v),("Std",s),("Min",np.min(a)),

# Build a Naive Bayes Spam Classifier
import numpy as np
def tk(t): return t.lower().split()
def train(emails,labels):
    sp,hm=[],[]
    for e,l in zip(emails,labels):
        w=tk(e)
        (sp if l else hm).extend(w)
    v=list(set(sp+hm))
    return ({w:(sp.count(w)+1)/(len(sp)+len(v)) for w in v},
            {w:(hm.count(w)+1)/(len(hm)+len(v)) for w in v},
            sum(labels)/len(labels))
def clf(email,sp,hm,ps):
    w=tk(email)
    ls=np.log(ps)+sum(np.log(sp.get(w,1e-10)) for w in w)
    lh=np.log(1-ps)+sum(np.log(hm.get(w,1e-10)) for w in w)
    return "SPAM" if ls>lh else "HAM"
ems=["buy cheap pills now","meeting at 3pm","win free money","project deadline","congrats you won","lunch at noon"]
lb=[1,0,1,0,1,0]
sp,hm,ps=train(ems,lb)
for t in ["free money","meeting tomorrow","cheap pills"]:
    print(f"'{t}' -> {clf(t,sp,hm,ps)}")

# Entropy in Decision Trees
import numpy as np
def entropy(labels):
    _, counts = np.unique(labels, return_counts=True)
    probs = counts / len(labels)
    return -np.sum(probs * np.log2(probs))
def gini(labels):
    _, counts = np.unique(labels, return_counts=True)
    probs = counts / len(labels)
    return 1 - np.sum(probs**2)
def info_gain(data, labels, split_col):
    parent_e = entropy(labels)
    values = np.unique(data[:, split_col])
    child_e = 0
    for v in values:
        mask = data[:, split_col] == v
        child_e += (sum(mask)/len(labels)) * entropy(labels[mask])
    return parent_e - child_e

for dist, desc in [([0,0,0,0,0], "All same"), ([0,0,0,0,1], "20%"), ([0,0,0,1,1], "40%"),

import numpy as np
import matplotlib.pyplot as plt

# Capstone: Implement linear regression with gradient descent from scratch
np.random.seed(42)
X = np.random.randn(200, 1)                          # 200 training examples
y = 3 * X.squeeze() + 2 + np.random.randn(200) * 0.5  # True: y = 3x + 2 + noise

# Initialise parameters
w, b = 0.0, 0.0
lr, n = 0.01, len(X)

losses = []
for epoch in range(1000):
    y_pred = w * X.squeeze() + b                      # Forward pass
    loss = np.mean((y_pred - y) ** 2)                 # MSE: L = (1/n)Σ(ŷ - y)²
    losses.append(loss)
    
    # Gradient computation: ∂L/∂w = (2/n)Σ(ŷ - y)·x,  ∂L/∂b = (2/n)Σ(ŷ - y)
    dw = (2 / n) * np.sum((y_pred - y) * X.squeeze())
    db = (2 / n) * np.sum(y_pred - y)
    
    # Gradient descent update: θ ← θ - α∇L
    w -= lr * dw
    b -= lr * db

print(f"Converged: w={w:.3f} (true: 3.0), b={b:.3f} (true: 2.0)")
print(f"Final MSE: {losses[-1]:.4f}")
assert abs(w - 3.0) < 0.15, f"w should be ~3, got {w:.3f}"
assert abs(b - 2.0) < 0.15, f"b should be ~2, got {b:.3f}"
