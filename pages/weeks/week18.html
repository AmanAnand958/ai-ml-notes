python — k8s_health_monitor.py
from typing import Dict, Any

class ModelHealthChecker:
    def __init__(self, model_loaded: bool, memory_usage_mb: float, memory_limit_mb: float = 4096.0):
        self.model_loaded = model_loaded
        self.memory_usage_mb = memory_usage_mb
        self.memory_limit_mb = memory_limit_mb

    def evaluate_readiness(self) -> Dict[str, Any]:
        if not self.model_loaded:
            return {"status": "UNREADY", "code": 503, "reason": "Model weights still loading into memory"}
        if self.memory_usage_mb > (self.memory_limit_mb * 0.95):
            return {"status": "DEGRADED", "code": 500, "reason": "Near OOM memory threshold"}
        return {"status": "HEALTHY", "code": 200, "memory_pct": round(self.memory_usage_mb / self.memory_limit_mb * 100, 2)}

checker = ModelHealthChecker(model_loaded=True, memory_usage_mb=1850.0)
print("Kubernetes Probe Response:", checker.evaluate_readiness())

What is a Kubernetes Pod?
The smallest deployable compute unit in K8s, containing one or more containers sharing network namespaces and storage volumes.
Deployment vs StatefulSet?
Deployments manage stateless replica pods (e.g. stateless ML APIs). StatefulSets manage stateful workloads requiring persistent identities (e.g. databases, vector stores).
Kubernetes Cluster IP vs NodePort vs LoadBalancer?
ClusterIP: internal cluster only. NodePort: exposes port on each node IP. LoadBalancer: provisions an external cloud load balancer to route traffic.
What is a K8s Ingress Controller?
An API object that manages external access to services, typically HTTP/HTTPS routing, SSL termination, and host/path-based load balancing.
Senior Interview Deep Dive: What trade-off occurs when scaling Kubernetes Basics for ML Deployments in distributed systems?
Trade-off balances inter-node communication latency vs compute saturation, resolved using overlap of computation and communication (ZeRO / DDP Ring-AllReduce).
⚠️ Gotcha: Missing Resource Requests Leading to Node OOM Kills

If you don't define `resources.requests` and `resources.limits` for memory and CPU in your ML Pod manifest, a single heavy batch prediction can consume all RAM on the node, causing the Linux OOM-killer to terminate the kubelet and crash the entire node.

import os

def initialize_service(model_name: str, cache_dir: str = "/tmp/models") -> dict:
    os.makedirs(cache_dir, exist_ok=True)
    weight_path = os.path.join(cache_dir, f"{model_name}.bin")

    return {"model": model_name, "path": weight_path, "status": "LOADED"}

from typing import Dict, Any

class ExperimentLogger:
    def __init__(self, experiment_name: str):
        self.experiment_name = experiment_name
        self.runs = []

    def log_run(self, params: Dict[str, Any], metrics: Dict[str, float], model_tag: str) -> Dict[str, Any]:
        run_data = {
            "run_id": f"run_{len(self.runs) + 1:04d}",
            "params": params,
            "metrics": metrics,
            "tag": model_tag,
            "status": "FINISHED"
        }
        self.runs.append(run_data)
        return run_data

import numpy as np

def compute_classification_metrics(y_true: np.ndarray, y_pred: np.ndarray) -> dict:
    tp = np.sum((y_true == 1) & (y_pred == 1))
    fp = np.sum((y_true == 0) & (y_pred == 1))
    fn = np.sum((y_true == 1) & (y_pred == 0))
    precision = tp / (tp + fp + 1e-9)
    recall = tp / (tp + fn + 1e-9)
    f1 = 2 * (precision * recall) / (precision + recall + 1e-9)
    return {"precision": round(float(precision), 4), "recall": round(float(recall), 4), "f1": round(float(f1), 4)}

    def forward(self, x):
        return self.net(x)

def train_epoch(model, dataloader, optimizer, criterion, device):
    model.train()
    total_loss, correct, total = 0.0, 0, 0
    for X_batch, y_batch in dataloader:
        X_batch, y_batch = X_batch.to(device), y_batch.to(device)
        optimizer.zero_grad()
        logits = model(X_batch)
        loss = criterion(logits, y_batch)
        loss.backward()
        nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
        optimizer.step()
        
        total_loss += loss.item() * X_batch.size(0)
        correct += (logits.argmax(dim=1) == y_batch).sum().item()
        total += y_batch.size(0)
        
    return total_loss / total, correct / total

🧠 Theory & Concepts

1. Core Mathematical Derivations: Cross-Entropy & Softmax

The Cross-Entropy loss for multi-class classification with true distribution $y$ and predicted probabilities $\hat{y} = \text{Softmax}(z)$ is:

$$\mathcal{L}_{\text{CE}} = -\sum_{i=1}^C y_i \log(\hat{y}_i) \quad \text{where } \hat{y}_i = \frac{e^{z_i}}{\sum_j e^{z_j}}$$ $$\frac{\partial \mathcal{L}}{\partial z_i} = \hat{y}_i - y_i \quad \text{(Gradient is simply predicted prob minus true label!)}$$

2. Vectorized Python Implementation (Live Coding Question)

import numpy as np

def softmax(logits: np.ndarray) -> np.ndarray:
    e_x = np.exp(logits - np.max(logits, axis=-1, keepdims=True))
    return e_x / np.sum(e_x, axis=-1, keepdims=True)

def cross_entropy_loss_and_grad(logits: np.ndarray, target_idx: int):
    probs = softmax(logits)
    loss = -np.log(probs[target_idx] + 1e-12)
    grad = probs.copy()
    grad[target_idx] -= 1.0 # dL/dz = p_i - y_i
    return loss, grad

logits = np.array([2.0, 1.0, 0.1])

        ],
        temperature=0,
    )

import math

def calculate_system_design_budget(
    qps=500,                    # Queries per second
    input_tokens=1024,          # Average prompt length
    output_tokens=128,          # Average response length
    model_params_b=8.0,         # 8 Billion parameter model
    precision_bytes=2,          # FP16 = 2 bytes, INT4 = 0.5 bytes
    kv_cache_heads=8,
    head_dim=128,
    num_layers=32
):
    # 1. Model Weights VRAM
    model_vram_gb = (model_params_b * 10**9 * precision_bytes) / (1024**3)
    
    # 2. KV Cache per concurrent request
    kv_bytes_per_token = 2 * num_layers * kv_cache_heads * head_dim * precision_bytes
    total_tokens_per_req = input_tokens + output_tokens
    kv_cache_per_req_mb = (kv_bytes_per_token * total_tokens_per_req) / (1024**2)
    
    # 3. Peak Concurrent Active Batches
    avg_latency_sec = 0.8
    concurrent_requests = qps * avg_latency_sec
    total_kv_vram_gb = (concurrent_requests * kv_cache_per_req_mb) / 1024
    
    total_vram_required = model_vram_gb + total_kv_vram_gb
    gpus_needed = math.ceil(total_vram_required / 80) # 80GB A100/H100 GPUs
    
    print(f"ML System Capacity Analysis (Target: {qps} QPS):")
    print(f"─" * 50)
    print(f"Model VRAM (Weights):       {model_vram_gb:.2f} GB")
    print(f"KV Cache VRAM ({concurrent_requests:.0f} conc): {total_kv_vram_gb:.2f} GB")
    print(f"Total VRAM Required:        {total_vram_required:.2f} GB")
    print(f"Minimum 80GB GPUs Needed:   {gpus_needed}x 80GB H100 SXM5")

import numpy as np

def compute_classification_metrics(y_true, y_pred_probs, threshold=0.5):
    y_pred = (y_pred_probs >= threshold).astype(int)
    tp = np.sum((y_true == 1) & (y_pred == 1))
    fp = np.sum((y_true == 0) & (y_pred == 1))
    fn = np.sum((y_true == 1) & (y_pred == 0))
    tn = np.sum((y_true == 0) & (y_pred == 0))
    
    accuracy = (tp + tn) / len(y_true)
    precision = tp / (tp + fp) if (tp + fp) > 0 else 0
    recall = tp / (tp + fn) if (tp + fn) > 0 else 0
    f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0
    
    return {"accuracy": accuracy, "precision": precision, "recall": recall, "f1_score": f1}
