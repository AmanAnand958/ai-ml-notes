import numpy as np

class PagedBlockManager:
    def __init__(self, block_size: int = 16, num_blocks: int = 100):
        self.block_size = block_size
        self.free_blocks = list(range(num_blocks))
        self.block_tables = {}
        self.ref_counts = {i: 0 for i in range(num_blocks)}

def allocate(self, req_id: str, num_tokens: int):
        num_blocks_needed = (num_tokens + self.block_size - 1) // self.block_size
        allocated = []
        for _ in range(num_blocks_needed):
            blk = self.free_blocks.pop(0)
            self.ref_counts[blk] += 1
            allocated.append(blk)
        self.block_tables[req_id] = allocated
return allocated

🧠 Theory & Concepts

1. GPU Memory Hierarchy & FlashAttention Tiling

Standard attention materializes the intermediate $N \times N$ attention matrix $S = Q K^T$ and $P = \text{softmax}(S)$ in slow GPU HBM ($1.5 - 3.0\text{ TB/s}$). FlashAttention partitions inputs into blocks and performs incremental softmax normalization entirely in high-speed GPU SRAM ($19\text{ TB/s}$).

graph TD subgraph "Standard Attention (Memory Bound)" Q1["Q, K in HBM"] --> Write1["Write N x N Matrix to HBM"] Write1 --> Softmax["Compute Softmax from HBM"] Softmax --> Write2["Write N x N Probs to HBM"] Write2 --> Output1["Multiply with V in HBM"] end subgraph "FlashAttention (Compute Bound Tiling)" Q2["Block Qi in SRAM"] & K2["Block Kj in SRAM"] --> Fused["Fused Tiled Kernel in SRAM (Online Softmax + Accumulate)"] Fused --> Output2["Final Output directly written to HBM"] end
FlashAttention SRAM Tiling: Bypassing the O(N^2) memory bandwidth read/write bottleneck.

2. Speculative Decoding Verification Principle

A lightweight Draft Model (e.g. 1B) predicts $K$ speculative tokens, and the Target Model (e.g. 70B) validates all $K$ tokens simultaneously in a single forward pass.

import numpy as np

def speculative_acceptance_step(draft_probs: np.ndarray, target_probs: np.ndarray, tokens: list):
    accepted = []
    for i, token in enumerate(tokens):
        p_draft = draft_probs[i]
        p_target = target_probs[i]
        ratio = p_target / max(1e-5, p_draft)
        if ratio >= 1.0 or np.random.uniform(0, 1) > 0.5:
            # Resample from residual distribution
            resampled = max(0, token - 1)
            accepted.append(resampled)
break
return accepted

def quantize_to_int8(weights: np.ndarray):
    max_val = np.max(np.abs(weights))
    scale = max_val / 127.0
    q_weights = np.clip(np.round(weights / (scale + 1e-9)), -128, 127).astype(np.int8)

import numpy as np

def online_softmax_tile(blocks: list):
    m_prev = -np.inf
    l_prev = 0.0
    out = np.zeros_like(blocks[0])
    
    for block in blocks:
        m_curr = np.maximum(m_prev, np.max(block))
        p = np.exp(block - m_curr)
        l_curr = np.exp(m_prev - m_curr) * l_prev + np.sum(p)
        
        out = (np.exp(m_prev - m_curr) * out + p) / l_curr
        m_prev = m_curr
        l_prev = l_curr
return out

def column_parallel_linear(X: np.ndarray, W: np.ndarray, num_gpus: int = 2):
