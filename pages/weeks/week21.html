# Step 1: Algorithmic Implementation & Core Transformations
import numpy as np

class PagedBlockManager:
    def __init__(self, block_size: int = 16, num_blocks: int = 100):
        self.block_size = block_size
        self.free_blocks = list(range(num_blocks))
        self.block_tables = {}
        self.ref_counts = {i: 0 for i in range(num_blocks)}

def allocate(self, req_id: str, num_tokens: int):
        num_blocks_needed = (num_tokens + self.block_size - 1) // self.block_size
        allocated = []
        for _ in range(num_blocks_needed):
            blk = self.free_blocks.pop(0)
            self.ref_counts[blk] += 1
            allocated.append(blk)
        self.block_tables[req_id] = allocated
return allocated

mgr = PagedBlockManager(block_size=16, num_blocks=50)
b = mgr.allocate("req_101", 38)
print(f"Allocated {len(b)} blocks for 38 tokens (Blocks: {b}). Remaining free: {len(mgr.free_blocks)}")

import numpy as np

def speculative_acceptance_step(draft_probs: np.ndarray, target_probs: np.ndarray, tokens: list):
    accepted = []
    for i, token in enumerate(tokens):
        p_draft = draft_probs[i]
        p_target = target_probs[i]
        ratio = p_target / max(1e-5, p_draft)
        if ratio >= 1.0 or np.random.uniform(0, 1) > 0.5:
            # Resample from residual distribution
            resampled = max(0, token - 1)
            accepted.append(resampled)
break
return accepted

np.random.seed(42)
acc = speculative_acceptance_step(np.array([0.8, 0.7, 0.4]), np.array([0.9, 0.75, 0.2]), [105, 204, 308])

def compute_speculative_speedup(alpha: float, gamma: int, c: float) -> float:
    expected_accepted = (1.0 - alpha ** (gamma + 1)) / (1.0 - alpha)
    cost = gamma * c + 1.0
return round(expected_accepted / cost, 2)

import numpy as np

def find_awq_salient_channels(weights: np.ndarray, activations: np.ndarray, top_k_percent: float = 0.05):
    # Activation magnitude per channel
    act_scale = np.mean(np.abs(activations), axis=0)
    num_salient = max(1, int(len(act_scale) * top_k_percent))
    salient_indices = np.argsort(act_scale)[-num_salient:]
    
    # Scale salient channels
    scaled_weights = weights.copy()
    scaled_weights[:, salient_indices] *= 2.0
return salient_indices, scaled_weights

def calculate_lora_trainable_params(base_params_m=7000, lora_r=16, d_model=4096, n_layers=32):
    # LoRA adds 2 * r * d_model parameters per adapted linear layer

# Step 1: Algorithmic Implementation & Core Transformations
import numpy as np

def online_softmax_tile(blocks: list):
    m_prev = -np.inf
    l_prev = 0.0
    out = np.zeros_like(blocks[0])
    
    for block in blocks:
        m_curr = np.maximum(m_prev, np.max(block))
        p = np.exp(block - m_curr)
        l_curr = np.exp(m_prev - m_curr) * l_prev + np.sum(p)
        
        out = (np.exp(m_prev - m_curr) * out + p) / l_curr
        m_prev = m_curr
        l_prev = l_curr
return out

class RadixPrefixNode:
def __init__(self):
        self.children = {}
        self.kv_cache_id = None
class RadixPrefixTree:
def __init__(self):
        self.root = RadixPrefixNode()

def insert(self, tokens: tuple, kv_id: str):
        curr = self.root
        for token in tokens:
            if token not in curr.children:
                curr.children[token] = RadixPrefixNode()
            curr = curr.children[token]
        curr.kv_cache_id = kv_id

def match_prefix(self, tokens: tuple):
        curr = self.root
        matched_len = 0
        last_kv = None
for token in tokens:
            if token in curr.children:
                curr = curr.children[token]
                matched_len += 1
if curr.kv_cache_id:
                    last_kv = curr.kv_cache_id
            else:
break
return matched_len, last_kv

def column_parallel_linear(X: np.ndarray, W: np.ndarray, num_gpus: int = 2):
    # Split W along columns
    W_splits = np.split(W, num_gpus, axis=1)

QUESTION 1 OF 4
In production LLM serving, what does TTFT (Time to First Token) measure?
A The latency from when a user request is received until the first generation token is emitted (dominated by prompt prefill processing).
B The time it takes to train the model from scratch.
C The time taken to download the model weights from the internet.
D The total duration of the entire user conversation.
✅ Correct! TTFT measures the prefill phase latency where the model processes all input prompt tokens in parallel to generate the first token.
❌ Incorrect. TTFT evaluates prompt ingestion and prefill latency, representing user perceived responsiveness before streaming begins.
QUESTION 2 OF 4
What does ITL (Inter-Token Latency) or TPOT (Time Per Output Token) quantify in streaming applications?
A The time elapsed between generating consecutive output tokens during the autoregressive decoding phase.
B The time taken to boot up the cloud server.
C The duration of network packet transmission over Wi-Fi.
D The time required to parse JSON outputs.
✅ Correct! ITL measures the speed of autoregressive decoding (tokens per second per stream), determining how smooth the streaming output feels to human readers.
❌ Incorrect. ITL / TPOT measures the per-token decoding latency during generation.
QUESTION 3 OF 4
Why does Prefix Caching (Prompt Caching) drastically improve throughput in multi-turn chatbot systems?
A It reuses the precomputed KV-cache for shared system prompts and conversation history across turns, avoiding redundant prefill computation.
B It deletes the user's past messages to save disk space.
C It converts conversation text into compressed MP3 audio.
D It disables attention mechanisms entirely.
✅ Correct! Reusing KV-cache for common prompt prefixes slashes TTFT by up to 90% and frees GPU compute for active generation.
❌ Incorrect. Prefix caching stores and reuses precomputed KV-tensors for common prompt prefixes, skipping redundant prefill passes.
QUESTION 4 OF 4
When designing an autoscaling policy for high-concurrency LLM inference clusters, what metric is most effective?
A vLLM waiting queue length (`num_requests_waiting`) and GPU KV-cache usage percentage.
B CPU core utilization only.
C Hard drive read/write speed.
D Number of open browser tabs.
✅ Correct! Scaling based on queue depth and KV-cache saturation directly addresses pending request backlogs before latency SLA violations occur.
❌ Incorrect. CPU usage is misleading for GPU inference; queue length (`num_requests_waiting`) and KV-cache allocation provide direct load signals.
